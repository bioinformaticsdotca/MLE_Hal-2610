import numpy as np
import matplotlib.pyplot as plt

from sklearn.datasets import make_moons
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.neural_network import MLPClassifier
from sklearn.metrics import accuracy_score
X, y = make_moons(
    n_samples=1000,
    noise=0.2,
    random_state=42
)

X_train, X_test, y_train, y_test = train_test_split(
    X,
    y,
    test_size=0.2,
    stratify=y,
    random_state=42
)

scaler = StandardScaler()

X_train = scaler.fit_transform(X_train)
X_test = scaler.transform(X_test)

Full-batch Gradient Descent

batch_size = len(X_train)

One subtle point: in scikit-learn’s implementation, the output activation is handled internally according to the classification task. activation=“relu” only specifies the hidden layers.

mlp_full = MLPClassifier(
    hidden_layer_sizes=(20, 20),
    activation="relu",
    solver="sgd",
    learning_rate_init=0.01,
    batch_size=len(X_train),   # Full batch
    max_iter=100,
    random_state=42
)

mlp_full.fit(X_train, y_train)
